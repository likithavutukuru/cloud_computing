from fastapi import FastAPI

app = FastAPI()

@app.get("/")
def home():
    return {"message": "API is running successfully"}

@app.get("/add")
def add(a: int, b: int):
    return {"result": a + b}

uvicorn main:app --host 0.0.0.0 --port 8000
